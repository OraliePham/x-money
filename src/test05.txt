  private initTables(): void {
    this.db.exec(`
      CREATE TABLE IF NOT EXISTS profiles (
        id TEXT PRIMARY KEY,
        username TEXT NOT NULL,
        display_name TEXT NOT NULL,
        email TEXT,
        user_id TEXT,
        cookies_json TEXT NOT NULL,

        metadata_json TEXT
      )
    `);

      CREATE TABLE IF NOT EXISTS sessions (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        profile_id TEXT NOT NULL,
        login_time DATETIME DEFAULT CURRENT_TIMESTAMP,
        logout_time DATETIME,
        ip_address TEXT,
        user_agent TEXT,
        session_duration INTEGER,
        status TEXT DEFAULT 'active',
        FOREIGN KEY (profile_id) REFERENCES profiles(id) ON DELETE CASCADE
      )

      CREATE TABLE IF NOT EXISTS tweets_cache (
        id TEXT PRIMARY KEY,

        id INTEGER PRIMARY KEY AUTOINCREMENT,
        profile_id TEXT NOT NULL,
        action_type TEXT NOT NULL,
        action_details TEXT,
        target_url TEXT,
        success INTEGER DEFAULT 1,
        error_message TEXT,
        created_at DATETIME DEFAULT CURRENT_TIMESTAMP

    this.db.exec(`
      CREATE INDEX IF NOT EXISTS idx_profiles_username ON profiles(username);

      stmt.run(now.toISOString(), duration, sessionId, profileId);
    }

import { Page } from 'playwright';

export class HumanBehavior {
  
  private static bezierCurve(t: number, p0: number, p1: number, p2: number, p3: number): number {
    const mt = 1 - t;
    return mt * mt * mt * p0 + 3 * mt * mt * t * p1 + 3 * mt * t * t * p2 + t * t * t * p3;
  }

  static async mouseMove(page: Page, targetX: number, targetY: number): Promise<void> {
    const start = await page.mouse.getPosition();
    const steps = 40 + Math.floor(Math.random() * 40);
    
    const cp1x = start.x + (targetX - start.x) * 0.3 + (Math.random() - 0.5) * 50;
    const cp1y = start.y + (targetY - start.y) * 0.3 + (Math.random() - 0.5) * 50;
    const cp2x = start.x + (targetX - start.x) * 0.7 + (Math.random() - 0.5) * 50;
    const cp2y = start.y + (targetY - start.y) * 0.7 + (Math.random() - 0.5) * 50;
    
    for (let i = 0; i <= steps; i++) {
      const t = i / steps;
      const x = this.bezierCurve(t, start.x, cp1x, cp2x, targetX);
      const y = this.bezierCurve(t, start.y, cp1y, cp2y, targetY);
      await page.mouse.move(x, y);
      await page.waitForTimeout(Math.random() * 8 + 3);
    }
  }

  static async type(page: Page, selector: string, text: string): Promise<void> {
    await page.click(selector);
    await page.waitForTimeout(Math.random() * 200 + 100);
    
    for (const char of text) {
      await page.keyboard.type(char);
      const delay = Math.random() * 80 + 40 + (char === ' ' ? 20 : 0);
      await page.waitForTimeout(delay);
    }
  }

  static async clickLikeHuman(page: Page, selector: string): Promise<void> {
    const element = await page.$(selector);
    if (element) {
      const boundingBox = await element.boundingBox();
      if (boundingBox) {
        const offsetX = boundingBox.width * (0.3 + Math.random() * 0.4);
        const offsetY = boundingBox.height * (0.3 + Math.random() * 0.4);
        await this.mouseMove(page, boundingBox.x + offsetX, boundingBox.y + offsetY);
        await this.delay(100, 300);
        await page.mouse.click(boundingBox.x + offsetX, boundingBox.y + offsetY);
      }
    } else {
      await page.click(selector);
    }
  }

  static async scroll(page: Page, distance: number): Promise<void> {
    const steps = 20 + Math.floor(Math.random() * 15);
    const stepDistance = distance / steps;
    let remaining = distance;
    
    for (let i = 0; i < steps && remaining > 0; i++) {
      const step = Math.min(stepDistance + (Math.random() - 0.5) * 20, remaining);
      await page.mouse.wheel(0, step);
      remaining -= step;
      await page.waitForTimeout(Math.random() * 50 + 25);
    }
  }

  static async delay(minMs: number = 500, maxMs: number = 2000): Promise<void> {
    const delay = Math.random() * (maxMs - minMs) + minMs;
    await new Promise(resolve => setTimeout(resolve, delay));
  }

  static async randomMouseMovement(page: Page): Promise<void> {
    const viewport = page.viewportSize();
    if (viewport) {
      const x = Math.random() * viewport.width;
      const y = Math.random() * viewport.height;
      await this.mouseMove(page, x, y);
    }
  }

  static async scrollToElement(page: Page, selector: string): Promise<void> {
    const element = await page.$(selector);
    if (element) {
      const boundingBox = await element.boundingBox();
      if (boundingBox) {
        await this.scroll(page, boundingBox.y);
        await this.delay(500, 1000);
      }
    }
  }

  static async hover(page: Page, selector: string): Promise<void> {
    const element = await page.$(selector);
    if (element) {
      const boundingBox = await element.boundingBox();
      if (boundingBox) {
        await this.mouseMove(page, boundingBox.x + boundingBox.width / 2, boundingBox.y + boundingBox.height / 2);
        await this.delay(100, 200);
      }
    }
  }
}

export class XActions {
  
  // ========== HELPER FUNCTIONS ==========
  
  static async waitForFeed(page: Page, timeout: number = 15000): Promise<boolean> {
    try {
      await page.waitForSelector('article[data-testid="tweet"]', { timeout });
      await HumanBehavior.delay(500, 1000);
      return true;
    } catch {
      console.warn(`Timeout waiting for feed after ${timeout}ms`);
      return false;
    }
  }

  static async getFirstTweetSafe(page: Page, timeout: number = 10000): Promise<TweetInfo | null> {
    try {
      const loaded = await this.waitForFeed(page, timeout);
      if (!loaded) return null;
      return await this.getFirstTweet(page);
    } catch (error) {
      console.warn('Failed to get first tweet:', error);
      return null;
    }
  }

  // ========== READ TWEETS ==========
  
  static async getFirstTweet(page: Page): Promise<TweetInfo | null> {
    await this.waitForFeed(page);
    
    return await page.evaluate(() => {
      const article = document.querySelector('article[data-testid="tweet"]');
      if (!article) return null;
      
      const textEl = article.querySelector('[data-testid="tweetText"]');
      const authorNameEl = article.querySelector('[data-testid="User-Name"] a span span');
      const authorUsernameEl = article.querySelector('[data-testid="User-Name"] a[href*="/"]:last-child span');
      const tweetLink = article.querySelector('a[href*="/status/"]')?.getAttribute('href');
      const timeEl = article.querySelector('time');
      const verifiedIcon = article.querySelector('[data-testid="icon-verified"]');
      
      const statsGroup = article.querySelector('div[role="group"][aria-label]');
      const ariaLabel = statsGroup?.getAttribute('aria-label') || '';
      
      const parseStat = (label: string, name: string): string => {
        const match = label.match(new RegExp(`(\\d+(?:,\\d+)*(?:\\.\\d+)?[KkM]?)\\s*${name}`));
        return match ? match[1] : '0';
      };
      
      return {
        text: textEl?.innerText || '',
        authorName: authorNameEl?.innerText || '',
        authorUsername: authorUsernameEl?.innerText?.replace('@', '') || '',
        tweetUrl: tweetLink ? `https://x.com${tweetLink}` : '',
        timestamp: timeEl?.getAttribute('datetime') || '',
        likes: parseStat(ariaLabel, 'likes?'),
        retweets: parseStat(ariaLabel, 'reposts?'),
        replies: parseStat(ariaLabel, 'replies?'),
        views: parseStat(ariaLabel, 'views?'),
        isVerified: !!verifiedIcon
      };
    });
  }

  static async getAllTweets(page: Page, maxTweets: number = 20): Promise<TweetInfo[]> {
    await this.waitForFeed(page);
    
    return await page.evaluate((max) => {
      const articles = Array.from(document.querySelectorAll('article[data-testid="tweet"]')).slice(0, max);
      
      return articles.map(article => {
        const textEl = article.querySelector('[data-testid="tweetText"]');
        const authorNameEl = article.querySelector('[data-testid="User-Name"] a span span');
        const authorUsernameEl = article.querySelector('[data-testid="User-Name"] a[href*="/"]:last-child span');
        const tweetLink = article.querySelector('a[href*="/status/"]')?.getAttribute('href');
        const timeEl = article.querySelector('time');
        
        return {
          text: textEl?.innerText || '',
          authorName: authorNameEl?.innerText || '',
          authorUsername: authorUsernameEl?.innerText?.replace('@', '') || '',
          tweetUrl: tweetLink ? `https://x.com${tweetLink}` : '',
          timestamp: timeEl?.getAttribute('datetime') || '',
        };
      });
    }, maxTweets);
  }

  // ========== CLICK TWEETS ==========
  
  static async clickFirstTweet(page: Page, humanLike: boolean = true): Promise<boolean> {
    const loaded = await this.waitForFeed(page);
    if (!loaded) return false;
    
    const tweet = await page.locator('article[data-testid="tweet"]').first();
    
    if (await tweet.count() === 0) return false;
    
    if (humanLike) {
      const box = await tweet.boundingBox();
      if (box) {
        await HumanBehavior.mouseMove(page, box.x + box.width / 2, box.y + box.height / 2);
        await HumanBehavior.delay(200, 400);
      }
    }
    
    await tweet.click();
    await page.waitForURL('**/status/**', { timeout: 10000 });
    await HumanBehavior.delay(1500, 2500);
    
    console.log('✅ Clicked first tweet');
    return true;
  }

  static async clickTweetByIndex(page: Page, index: number, humanLike: boolean = true): Promise<boolean> {
    const loaded = await this.waitForFeed(page);
    if (!loaded) return false;
    
    const tweets = await page.$$('article[data-testid="tweet"]');
    if (index >= tweets.length) return false;
    
    const tweet = tweets[index];
    
    if (humanLike) {
      const box = await tweet.boundingBox();
      if (box) {
        await HumanBehavior.mouseMove(page, box.x + box.width / 2, box.y + box.height / 2);
        await HumanBehavior.delay(200, 400);
      }

    await tweet.click();

    const likeButton = await page.locator('[data-testid="like"]').first();
    
    if (await likeButton.count() === 0) return false;
    
    const ariaLabel = await likeButton.getAttribute('aria-label');
    const isLiked = ariaLabel?.includes('Unlike');
    
    if (!isLiked) {
      await HumanBehavior.clickLikeHuman(page, '[data-testid="like"]');
      await HumanBehavior.delay(500, 1000);
      console.log('❤️ Liked tweet');
      return true;
    }
    
    console.log('Tweet already liked');
    return false;
  }

function normalizeUrl(rawUrl: string): string {
  const trimmed = rawUrl.trim();
  const repaired =
    trimmed.startsWith('https:/') && !trimmed.startsWith('https://')
      ? trimmed.replace('https:/', 'https://')
      : trimmed;

  return new URL(repaired).toString();
}

function isXUrl(targetUrl: string): boolean {

    console.log(`   💬 Text: ${firstTweet.text.substring(0, 100)}...`);
    console.log(`   ❤️ Likes: ${firstTweet.likes ?? '0'} | 🔁 Retweets: ${firstTweet.retweets ?? '0'}`);
  } else {
    console.log('   ⚠️ No tweets found in feed');
    console.log('\n' + '='.repeat(60));
    console.log('⚠️ X ACTIONS DEMO SKIPPED - No tweets available');
    console.log('='.repeat(60));
    return;
  }

  // Step 3: Scroll feed
  console.log('\n📌 Step 3: Scrolling feed...');
  await XActions.scrollFeed(page, 2);

  // Step 4: Get multiple tweets
  console.log('\n📌 Step 4: Getting multiple tweets...');
  const allTweets = await XActions.getAllTweets(page, 5);
  console.log(`   📊 Found ${allTweets.length} tweets in feed`);

  // Step 5: Click first tweet
  console.log('\n📌 Step 5: Opening first tweet detail...');
  try {
    const clicked = await XActions.clickFirstTweet(page, true);
    if (clicked) {
      console.log('   ✅ Opened tweet detail');

      console.log('\n📌 Step 6: Getting current tweet details...');
      const tweetDetail = await XActions.getCurrentTweetDetail(page);
      if (tweetDetail) {
        console.log(`   📝 Full text: ${tweetDetail.text.substring(0, 150)}...`);
      }

      console.log('\n📌 Step 7: Going back to home...');
      await XActions.goBack(page);
    } else {
      console.log('   ⚠️ Could not click first tweet');
    }
  } catch (error) {
    console.error('   ❌ Error clicking tweet:', error);
  }

  // Step 8: Check notifications
  console.log('\n📌 Step 8: Checking notifications...');
  const unreadCount = await XActions.getUnreadCount(page);
  console.log(`   🔔 Unread notifications: ${unreadCount}`);

  // Step 9: Get user profile (if username provided)
  if (profileUsername) {
    console.log('\n📌 Step 9: Getting current user profile info...');
    try {
      const userProfile = await XActions.getUserProfile(page, profileUsername);
      if (userProfile) {
        console.log(`   👤 @${userProfile.username}`);
        console.log(`   📝 Bio: ${userProfile.bio?.substring(0, 100) || 'No bio'}`);
        console.log(`   👥 Followers: ${userProfile.followersCount}`);
        console.log(`   Following: ${userProfile.followingCount}`);
      }
    } catch (error) {
      console.error('   ❌ Error getting user profile:', error);
    }
  }

  console.log('\n' + '='.repeat(60));
  console.log('✅ X ACTIONS DEMO COMPLETED');
  console.log('='.repeat(60));
}

async function likeCurrentTweetIfRequested(page: Page, shouldLike: boolean): Promise<void> {
  if (!shouldLike) return;

  console.log('\n❤️ likeCurrentTweet flag detected.');

  try {
    if (!page.url().includes('/status/')) {
      console.log('📌 Current page is not a tweet detail page. Opening first tweet from feed...');
      const clicked = await XActions.clickFirstTweet(page, true);
      if (!clicked) {
        console.log('❌ Could not open first tweet, skipping like.');
        return;
      }
    }

    const liked = await XActions.likeCurrentTweet(page);
    console.log(liked ? '✅ likeCurrentTweet completed.' : '⚠️ Tweet already liked or like button not found.');
  } catch (error) {
    console.error('❌ Failed to run likeCurrentTweet:', error);
  }
}

async function runOptionalXActions(
  page: Page,
  manager: ProfileManagerWithSQLite,
  profileId: string,
  args: LauncherArgs,
): Promise<void> {
  if (!isXUrl(args.targetUrl)) return;

  const refreshedProfile = manager.getStoredProfile(profileId);
  if (!refreshedProfile) {
    console.log('\n⚠️ Not logged into X yet. Skipping optional X actions.');
    console.log('💡 Please log in manually, then run again.');
    return;
  }

  await likeCurrentTweetIfRequested(page, args.likeCurrentTweet);

  if (args.demoActions) {
    await demoXActions(page, refreshedProfile.username);
  }

  if (args.demoSearch) {
    await demoSearchActions(page);
  }
}

async function main(): Promise<void> {
  const args = parseArgs(process.argv.slice(2));
  const manager = new ProfileManagerWithSQLite('./browser_profiles', DEFAULT_DB_PATH);
  const metadata = await manager.ensureProfile(args.profileId, args.targetUrl);
  const storedProfile = manager.getStoredProfile(args.profileId);

  console.log(`\n📦 Opening profile ${metadata.id}`);
  console.log(`📁 User data: ${metadata.userDataDir}`);
  console.log(`🌐 Target URL: ${args.targetUrl}`);
  console.log(`💾 SQLite DB: ${manager.getStorage().getDatabasePath()}`);
  if (storedProfile) {
    console.log(`📱 Stored X profile: @${storedProfile.username} (${storedProfile.displayName})`);
  }

  if (args.dryRun) {
    console.log('\n✅ Dry run completed. Browser launch skipped.');
    manager.getStorage().close();
    return;
  }

  console.log('\n🖱️ Close the browser window or press Ctrl+C to stop.\n');

  const { context, page } = await manager.launchProfileWithRestore(args.profileId, {
    targetUrl: args.targetUrl,
  });

  await runHumanBehaviorWarmup(page);
  await captureXProfileIfApplicable(manager, page, args.profileId, args.targetUrl);
  await runOptionalXActions(page, manager, args.profileId, args);

  console.log('\n💡 Browser will stay open. Press Ctrl+C or close the browser window to stop.');
  await context.waitForEvent('close');
  await manager.closeAll();
}

if (isDirectRun()) {
  main().catch((error: unknown) => {
    console.error('❌ Fatal error:', error);
    process.exitCode = 1;
  });
}
